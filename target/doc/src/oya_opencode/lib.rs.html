5//! # oya-opencode
6//!
7//! OpenCode AI bridge for OYA - Execute AI coding tasks via opencode CLI.
8//!
9//! This crate provides integration with the [OpenCode](https://opencode.ai) CLI
10//! for AI-powered code generation, analysis, and modification.
11//!
12//! ## Features
13//!
14//! - Execute prompts via the opencode CLI
15//! - Stream responses for real-time feedback
16//! - Phase-based execution with context management
17//! - Configurable models and agent modes
18//!
19//! ## Example
20//!
21//! ```ignore
22//! use oya_opencode::{OpencodeClient, PhaseContext, AIExecutor};
23//!
24//! // Create a client
25//! let client = OpencodeClient::new()?;
26//!
27//! // Execute a simple prompt
28//! let result = client.execute("Create a hello world function in Rust").await?;
29//! println!("Output: {}", result.output);
30//!
31//! // Or use phase-based execution
32//! let executor = AIExecutor::new(Arc::new(client));
33//! let ctx = PhaseContext::new("implement", "Create a new feature")
34//!     .with_input(PhaseInput::text("Build a CLI parser"))
35//!     .with_constraint("Use clap");
36//! let output = executor.execute(&ctx).await?;
37//! ```
