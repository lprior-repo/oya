1// Layout module - Terminal layout system for OYA UI
2//
3// Defines the 3-pane layout structure and pane types for the plugin.
4
5use serde::{Deserialize, Serialize};
6use std::fmt;
7use thiserror::Error;
8
9/// Errors that can occur in layout operations
10#[derive(Debug, Error)]
11pub enum LayoutError {
12    #[error("Invalid pane dimensions: {0}")]
13    InvalidDimensions(String),
14
15    #[error("Pane not found: {0}")]
16    PaneNotFound(String),
17
18    #[error("Layout calculation error: {0}")]
19    CalculationError(String),
20}
21
22/// Result type for layout operations
23pub type LayoutResult<T> = Result<T, LayoutError>;
24
25/// Types of panes in the OYA UI layout
26#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
27pub enum PaneType {
28    /// Left pane showing list of beads
29    BeadList,
30    /// Right top pane showing bead details
31    BeadDetail,
32    /// Right top pane showing pipeline progress
33    PipelineView,
34    /// Bottom pane showing workflow graph
35    WorkflowGraph,
36}
37
38impl fmt::Display for PaneType {
39    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
40        match self {
41            PaneType::BeadList => write!(f, "Bead List"),
42            PaneType::BeadDetail => write!(f, "Bead Detail"),
43            PaneType::PipelineView => write!(f, "Pipeline View"),
44            PaneType::WorkflowGraph => write!(f, "Workflow Graph"),
45        }
46    }
47}

49/// A pane in the terminal layout
50#[derive(Debug, Clone, Serialize, Deserialize)]
51pub struct Pane {
52    /// Pane type
53    pub pane_type: PaneType,
54    /// Row position (0-indexed)
55    pub row: usize,
56    /// Column position (0-indexed)
57    pub col: usize,
58    /// Height in rows
59    pub height: usize,
60    /// Width in columns
61    pub width: usize,
62    /// Pane title
63    pub title: String,
64}
65
66impl Pane {
67    /// Create a new pane
68    ///
69    /// # Arguments
70    ///
71    /// * `pane_type` - Type of pane
72    /// * `row` - Row position
73    /// * `col` - Column position
74    /// * `height` - Height in rows
75    /// * `width` - Width in columns
76    ///
77    /// # Errors
78    ///
79    /// Returns an error if dimensions are invalid
80    pub fn new(
81        pane_type: PaneType,
82        row: usize,
83        col: usize,
84        height: usize,
85        width: usize,
86    ) -> LayoutResult<Self> {
87        if height == 0 || width == 0 {
88            return Err(LayoutError::InvalidDimensions(
89                "Pane dimensions must be positive".to_string(),
90            ));
91        }
92
93        let title = match pane_type {
94            PaneType::BeadList => "Beads",
95            PaneType::BeadDetail => "Details",
96            PaneType::PipelineView => "Pipeline",
97            PaneType::WorkflowGraph => "Workflow",
98        }
99        .to_string();
100
101        Ok(Self {
102            pane_type,
103            row,
104            col,
105            height,
106            width,
107            title,
108        })
109    }
110
111    /// Create a new pane with hardcoded default values (internal use only)
112    ///
113    /// # Note
114    ///
115    /// This bypasses validation since the hardcoded defaults are known to be valid.
116    /// Only use with hardcoded constants that have been verified.
117    #[must_use]
118    fn with_defaults(
119        pane_type: PaneType,
120        row: usize,
121        col: usize,
122        height: usize,
123        width: usize,
124    ) -> Self {
125        let title = match pane_type {
126            PaneType::BeadList => "Beads",
127            PaneType::BeadDetail => "Details",
128            PaneType::PipelineView => "Pipeline",
129            PaneType::WorkflowGraph => "Workflow",
130        };
131
132        Self {
133            pane_type,
134            row,
135            col,
136            height,
137            width,
138            title: title.to_string(),
139        }
140    }
141
142    /// Get the right boundary column
143    #[must_use]
144    pub const fn right(&self) -> usize {
145        self.col + self.width
146    }
147
148    /// Get the bottom boundary row
149    #[must_use]
150    pub const fn bottom(&self) -> usize {
151        self.row + self.height
152    }
153}

155/// Terminal layout configuration
156///
157/// Implements a 3-pane layout:
158/// ┌─────────────────────────────────┐
159/// │ BeadList      │ BeadDetail       │
160/// │               ├─────────────────┤
161/// │               │ PipelineView     │
162/// ├───────────────┴─────────────────┤
163/// │ WorkflowGraph                   │
164/// └─────────────────────────────────┘
165#[derive(Debug, Clone, Serialize, Deserialize)]
166pub struct Layout {
167    /// All panes in the layout
168    panes: Vec<Pane>,
169}
170
171impl Layout {
172    /// Create a new 3-pane layout with default dimensions
173    ///
174    /// # Returns
175    ///
176    /// A new layout with default 3-pane configuration (80x24 terminal)
177    #[must_use]
178    pub fn new_3_pane() -> Self {
179        let bead_list = Pane::with_defaults(PaneType::BeadList, 1, 1, 15, 32);
180        let bead_detail = Pane::with_defaults(PaneType::BeadDetail, 1, 34, 8, 45);
181        let pipeline_view = Pane::with_defaults(PaneType::PipelineView, 10, 34, 6, 45);
182        let workflow_graph = Pane::with_defaults(PaneType::WorkflowGraph, 17, 1, 6, 78);
183
184        Self {
185            panes: vec![bead_list, bead_detail, pipeline_view, workflow_graph],
186        }
187    }
188
189    /// Get all panes in the layout
190    #[must_use]
191    pub const fn panes(&self) -> &Vec<Pane> {
192        &self.panes
193    }
194
195    /// Get a pane by type
196    ///
197    /// # Arguments
198    ///
199    /// * `pane_type` - Type of pane to find
200    ///
201    /// # Returns
202    ///
203    /// Reference to the pane if found
204    pub fn get_pane(&self, pane_type: PaneType) -> Option<&Pane> {
205        self.panes.iter().find(|p| p.pane_type == pane_type)
206    }
207
208    /// Calculate layout for a given terminal size
209    ///
210    /// # Arguments
211    ///
212    /// * `rows` - Total rows in terminal
213    /// * `cols` - Total columns in terminal
214    ///
215    /// # Returns
216    ///
217    /// A new layout adjusted for the terminal size
218    ///
219    /// # Errors
220    ///
221    /// Returns an error if the terminal is too small or pane creation fails
222    pub fn calculate_for_terminal(rows: usize, cols: usize) -> LayoutResult<Self> {
223        if rows < 20 || cols < 40 {
224            return Err(LayoutError::InvalidDimensions(
225                "Terminal must be at least 20 rows x 40 columns".to_string(),
226            ));
227        }
228
229        // Calculate dimensions based on terminal size
230        // Left pane: 40% width
231        let left_width = cols.saturating_mul(40).saturating_div(100);
232        // Right panes: 60% width
233        let right_width = cols.saturating_sub(left_width).saturating_sub(3); // -3 for borders
234        // Top panes: 60% height
235        let top_height = rows.saturating_mul(60).saturating_div(100);
236        // Bottom pane: 40% height
237        let bottom_height = rows.saturating_sub(top_height).saturating_sub(3); // -3 for borders
238
239        // Split right top into bead detail (60%) and pipeline view (40%)
240        let bead_detail_height = top_height.saturating_mul(60).saturating_div(100);
241        let pipeline_height = top_height
242            .saturating_sub(bead_detail_height)
243            .saturating_sub(2);
244
245        // Validate calculated dimensions before creating panes
246        let bead_list_height = top_height.saturating_sub(1);
247        if bead_list_height == 0 || left_width == 0 {
248            return Err(LayoutError::CalculationError(
249                "Calculated bead list dimensions are invalid".to_string(),
250            ));
251        }
252
253        let bead_list = Pane::new(
254            PaneType::BeadList,
255            1,
256            1,
257            bead_list_height,
258            left_width,
259        )?;
260
261        if bead_detail_height == 0 || right_width == 0 {
262            return Err(LayoutError::CalculationError(
263                "Calculated bead detail dimensions are invalid".to_string(),
264            ));
265        }
266
267        let bead_detail = Pane::new(
268            PaneType::BeadDetail,
269            1,
270            left_width.saturating_add(3),
271            bead_detail_height,
272            right_width,
273        )?;
274
275        if pipeline_height == 0 {
276            return Err(LayoutError::CalculationError(
277                "Calculated pipeline view dimensions are invalid".to_string(),
278            ));
279        }
280
281        let pipeline_view = Pane::new(
282            PaneType::PipelineView,
283            bead_detail_height.saturating_add(3),
284            left_width.saturating_add(3),
285            pipeline_height,
286            right_width,
287        )?;
288
289        if bottom_height == 0 {
290            return Err(LayoutError::CalculationError(
291                "Calculated workflow graph dimensions are invalid".to_string(),
292            ));
293        }
294
295        let workflow_graph = Pane::new(
296            PaneType::WorkflowGraph,
297            top_height.saturating_add(3),
298            1,
299            bottom_height,
300            cols,
301        )?;
302
303        Ok(Self {
304            panes: vec![bead_list, bead_detail, pipeline_view, workflow_graph],
305        })
306    }
307
308    /// Validate the layout
309    ///
310    /// # Returns
311    ///
312    /// Ok if the layout is valid, Err otherwise
313    pub fn validate(&self) -> LayoutResult<()> {
314        for pane in &self.panes {
315            if pane.height == 0 || pane.width == 0 {
316                return Err(LayoutError::InvalidDimensions(format!(
317                    "Pane {} has invalid dimensions",
318                    pane.pane_type
319                )));
320            }
321        }
322        Ok(())
323    }
324}
325
326impl Default for Layout {
327    fn default() -> Self {
328        Self::new_3_pane()
329    }

332#[cfg(test)]
333mod tests {
334    use super::*;
335
336    #[test]
337    fn test_layout_creation() {
338        let layout = Layout::new_3_pane();
339        assert_eq!(layout.panes().len(), 4);
340    }
341
342    #[test]
343    fn test_get_pane() {
344        let layout = Layout::new_3_pane();
345        let bead_list = layout.get_pane(PaneType::BeadList);
346        assert!(bead_list.is_some());
347        if let Some(pane) = bead_list {
348            assert_eq!(pane.pane_type, PaneType::BeadList);
349        }
350    }
351
352    #[test]
353    fn test_calculate_for_terminal() {
354        let layout_result = Layout::calculate_for_terminal(24, 80);
355        assert!(layout_result.is_ok());
356        let layout = layout_result.unwrap_or_else(|_| Layout::new_3_pane());
357        assert_eq!(layout.panes().len(), 4);
358
359        let bead_list = layout.get_pane(PaneType::BeadList);
360        assert!(bead_list.is_some());
361        if let Some(pane) = bead_list {
362            assert_eq!(pane.width, 32); // 40% of 80
363        }
364
365        let workflow_graph = layout.get_pane(PaneType::WorkflowGraph);
366        assert!(workflow_graph.is_some());
367        if let Some(pane) = workflow_graph {
368            assert!(pane.height > 0);
369        }
370    }
371
372    #[test]
373    fn test_calculate_too_small() {
374        let result = Layout::calculate_for_terminal(10, 20);
375        assert!(result.is_err());
376    }
377
378    #[test]
379    fn test_pane_boundaries() {
380        let pane_result = Pane::new(PaneType::BeadList, 5, 10, 15, 30);
381        assert!(pane_result.is_ok());
382        let pane = pane_result.unwrap_or_else(|_| {
383            Pane::with_defaults(PaneType::BeadList, 5, 10, 15, 30)
384        });
385        assert_eq!(pane.right(), 40);
386        assert_eq!(pane.bottom(), 20);
387    }
388
389    #[test]
390    fn test_validate_layout() {
391        let layout = Layout::new_3_pane();
392        assert!(layout.validate().is_ok());
393    }
