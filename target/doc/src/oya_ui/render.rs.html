38    /// Render the complete layout
39    ///
40    /// # Arguments
41    ///
42    /// * `layout` - Layout configuration
43    /// * `beads` - List of beads to display
44    /// * `selected_index` - Index of selected bead
45    /// * `focused_pane` - Currently focused pane type
46    ///
47    /// # Returns
48    ///
49    /// Complete rendered output as a string
50    #[must_use]
51    pub fn render_layout(
52        &self,
53        layout: &Layout,
54        beads: &[SampleBead],
55        selected_index: usize,
56        focused_pane: PaneType,
57    ) -> String {
58        let mut output = String::new();
59
60        // Clear screen
61        output.push_str("\x1b[2J\x1b[H");
62
63        // Render each pane
64        for pane in layout.panes() {
65            let content = match pane.pane_type {
66                PaneType::BeadList => {
67                    self.render_bead_list(pane, beads, selected_index, focused_pane)
68                }
69                PaneType::BeadDetail => {
70                    self.render_bead_detail(pane, beads, selected_index, focused_pane)
71                }
72                PaneType::PipelineView => {
73                    self.render_pipeline_view(pane, beads, selected_index, focused_pane)
74                }
75                PaneType::WorkflowGraph => self.render_workflow_graph(pane, focused_pane),
76            };
77
78            // Render pane border and content
79            let pane_output = self.render_pane(pane, &content, focused_pane);
80            output.push_str(&pane_output);
81        }
82
83        // Render status bar at bottom
84        let status = self.render_status_bar(focused_pane);
85        output.push_str(&status);
86
87        output
88    }

90    /// Render a single pane with border and content
91    fn render_pane(&self, pane: &Pane, content: &str, focused_pane: PaneType) -> String {
92        let is_focused = pane.pane_type == focused_pane;
93        let mut output = String::new();
94
95        // Move cursor to pane position
96        write!(output, "\x1b[{};{}H", pane.row, pane.col).ok();
97
98        // Top border
99        if is_focused {
100            let title = style::colorize(&pane.title, style::COLOR_GREEN);
101            output.push_str(&self.render_top_border(pane.width, &title));
102        } else {
103            output.push_str(&self.render_top_border(pane.width, &pane.title));
104        }
105
106        // Content area
107        let content_lines: Vec<&str> = content.lines().collect();
108        let content_height = pane.height.saturating_sub(2); // Account for top and bottom borders
109
110        for i in 0..content_height {
111            write!(
112                output,
113                "\x1b[{};{}H",
114                pane.row.saturating_add(1).saturating_add(i),
115                pane.col
116            )
117            .ok();
118
119            let line = content_lines.get(i);
120            if let Some(line_text) = line {
121                output.push_str("│ ");
122                output.push_str(line_text);
123                output.push_str(
124                    &" ".repeat(
125                        pane.width
126                            .saturating_sub(2)
127                            .saturating_sub(line_text.chars().count()),
128                    ),
129                );
130                output.push('│');
131            } else {
132                output.push('│');
133                output.push_str(&" ".repeat(pane.width.saturating_sub(2)));
134                output.push('│');
135            }
136        }
137
138        // Bottom border
139        write!(
140            output,
141            "\x1b[{};{}H",
142            pane.row.saturating_add(pane.height).saturating_sub(1),
143            pane.col
144        )
145        .ok();
146        output.push_str(&self.render_bottom_border(pane.width));
147
148        output
149    }

151    /// Render top border with title
152    fn render_top_border(&self, width: usize, title: &str) -> String {
153        let mut output = String::from("┌");
154
155        // Add title (truncated if too long)
156        let title_len = title.chars().count();
157        let available_width = width.saturating_sub(4);
158
159        if title_len <= available_width {
160            output.push_str(title);
161            output.push_str(&"─".repeat(width.saturating_sub(2).saturating_sub(title_len)));
162        } else {
163            let truncated: String = title.chars().take(available_width).collect();
164            output.push_str(&truncated);
165            output.push_str(&"─".repeat(width.saturating_sub(2).saturating_sub(available_width)));
166        }
167
168        output.push('┐');
169        output.push('\n');
170        output
171    }

173    /// Render bottom border
174    fn render_bottom_border(&self, width: usize) -> String {
175        let mut output = String::from("└");
176        output.push_str(&"─".repeat(width.saturating_sub(2)));
177        output.push('┘');
178        output
179    }

181    /// Render bead list pane
182    fn render_bead_list(
183        &self,
184        pane: &Pane,
185        beads: &[SampleBead],
186        selected_index: usize,
187        focused_pane: PaneType,
188    ) -> String {
189        let is_focused = pane.pane_type == focused_pane;
190        let mut content = String::new();
191
192        if beads.is_empty() {
193            content.push_str("No beads available.");
194            return content;
195        }
196
197        // Header
198        if is_focused {
199            content.push_str(&style::colorize(
200                "   ID      Priority  State     Title",
201                style::COLOR_GREEN,
202            ));
203        } else {
204            content.push_str("   ID      Priority  State     Title");
205        }
206        content.push('\n');
207
208        // Beads
209        for (i, bead) in beads.iter().enumerate() {
210            if i >= pane.height.saturating_sub(3) {
211                break; // Don't overflow pane
212            }
213
214            let state_color = match bead.state.as_str() {
215                "open" => style::COLOR_RESET,
216                "in_progress" => style::COLOR_GREEN,
217                "blocked" => style::COLOR_RED,
218                "closed" => style::COLOR_RESET,
219                _ => style::COLOR_RESET,
220            };
221
222            let priority = match bead.priority {
223                1 => "P1",
224                2 => "P2",
225                3 => "P3",
226                _ => "P?",
227            };
228
229            let marker = if i == selected_index { "→" } else { " " };
230
231            let line = format!(
232                "{} {:8} {:8} {:9} {}",
233                marker,
234                truncate_bead_id(&bead.id, 8),
235                priority,
236                bead.state,
237                truncate(&bead.title, 20)
238            );
239
240            if is_focused {
241                content.push_str(&style::colorize(&line, state_color));
242            } else {
243                content.push_str(&line);
244            }
245            content.push('\n');
246        }
247
248        content
249    }

251    /// Render bead detail pane
252    fn render_bead_detail(
253        &self,
254        pane: &Pane,
255        beads: &[SampleBead],
256        selected_index: usize,
257        focused_pane: PaneType,
258    ) -> String {
259        let is_focused = pane.pane_type == focused_pane;
260        let mut content = String::new();
261
262        let bead = match beads.get(selected_index) {
263            Some(b) => b,
264            None => {
265                content.push_str("No bead selected.");
266                return content;
267            }
268        };
269
270        // Header
271        if is_focused {
272            content.push_str(&style::colorize(&bead.title, style::COLOR_GREEN));
273        } else {
274            content.push_str(&bead.title);
275        }
276        content.push('\n');
277        content.push('\n');
278
279        // Details
280        content.push_str(&format!("ID:          {}\n", bead.id));
281        content.push_str(&format!("Priority:    P{}\n", bead.priority));
282        content.push_str(&format!("State:       {}\n", bead.state));
283        content.push('\n');
284
285        // Description placeholder
286        content.push_str("Full bead details will be available\n");
287        content.push_str("after IPC integration is implemented.\n");
288
289        content
290    }

292    /// Render pipeline view pane
293    fn render_pipeline_view(
294        &self,
295        pane: &Pane,
296        beads: &[SampleBead],
297        selected_index: usize,
298        focused_pane: PaneType,
299    ) -> String {
300        let is_focused = pane.pane_type == focused_pane;
301        let mut content = String::new();
302
303        if beads.is_empty() || selected_index >= beads.len() {
304            content.push_str("Select a bead to view pipeline.");
305            return content;
306        }
307
308        // Header
309        if is_focused {
310            content.push_str(&style::colorize("Pipeline Stages", style::COLOR_GREEN));
311        } else {
312            content.push_str("Pipeline Stages");
313        }
314        content.push('\n');
315        content.push('\n');
316
317        // Stages (placeholder for now)
318        let stages = [
319            ("implement", "pending"),
320            ("unit-test", "pending"),
321            ("coverage", "pending"),
322            ("lint", "pending"),
323            ("static", "pending"),
324            ("integration", "pending"),
325            ("security", "pending"),
326            ("review", "pending"),
327        ];
328
329        for (stage, status) in stages {
330            let status_char = match status {
331                "pending" => "○",
332                "running" => "◐",
333                "complete" => "●",
334                "failed" => "✗",
335                _ => "?",
336            };
337
338            if is_focused && status == "running" {
339                content.push_str(&style::colorize(
340                    &format!("  {} {}", status_char, stage),
341                    style::COLOR_GREEN,
342                ));
343            } else {
344                content.push_str(&format!("  {} {}", status_char, stage));
345            }
346            content.push('\n');
347        }
348
349        content
350    }

404/// Truncate text to fit width
405fn truncate(text: &str, width: usize) -> String {
406    let chars: Vec<char> = text.chars().collect();
407    if chars.len() <= width {
408        text.to_string()
409    } else if width > 3 {
410        let truncated: String = chars.iter().take(width.saturating_sub(3)).collect();
411        format!("{}...", truncated)
412    } else {
413        "...".to_string()
414    }
415}

417/// Truncate bead ID to fit width (no ellipsis for IDs)
418fn truncate_bead_id(text: &str, width: usize) -> String {
419    let chars: Vec<char> = text.chars().collect();
420    if chars.len() <= width {
421        text.to_string()
422    } else {
423        chars.iter().take(width).collect()
424    }
425}

427/// Wrap text to fit width
428fn textwrap(text: &str, width: usize) -> Vec<String> {
429    let mut lines = Vec::new();
430    let mut current_line = String::new();
431    let mut current_length = 0;
432
433    for word in text.split_whitespace() {
434        let word_len = word.chars().count();
435
436        if current_length == 0 {
437            current_line.push_str(word);
438            current_length = word_len;
439        } else if current_length.saturating_add(1).saturating_add(word_len) <= width {
440            current_line.push(' ');
441            current_line.push_str(word);
442            current_length = current_length.saturating_add(1).saturating_add(word_len);
443        } else {
444            lines.push(current_line);
445            current_line = word.to_string();
446            current_length = word_len;
447        }
448    }
449
450    if !current_line.is_empty() {
451        lines.push(current_line);
452    }
453
454    lines
455}
